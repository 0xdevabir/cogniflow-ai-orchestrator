1. Get the application URL by running these commands:
{{- if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  {{- range .paths }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}
{{- end }}
{{- else if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "cogniflow.fullname" . }}-web)
  export NODE_IP=$(kubectl get nodes -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
  NOTE: It may take a few minutes for the LoadBalancer IP to be available.
        You can watch the status of by running 'kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "cogniflow.fullname" . }}-web'
  export SERVICE_IP=$(kubectl get svc {{ include "cogniflow.fullname" . }}-web --namespace {{ .Release.Namespace }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.web.port }}
{{- else if contains "ClusterIP" .Values.service.type }}
  export POD_NAME=$(kubectl get pods -l "app.kubernetes.io/name={{ include "cogniflow.name" . }},app.kubernetes.io/instance={{ .Release.Name }},app.kubernetes.io/component=web" -o jsonpath="{.items[0].metadata.name}")
  export CONTAINER_PORT=$(kubectl get pod $POD_NAME -o jsonpath="{.spec.containers[0].ports[0].containerPort}")
  echo "Visit http://127.0.0.1:8080 to use the web app"
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8080:$CONTAINER_PORT
{{- end }}

2. Check the orchestrator is healthy:
  kubectl get pods -l "app.kubernetes.io/component=orchestrator"
  kubectl logs -l "app.kubernetes.io/component=orchestrator" --tail=20

3. (Optional) Wire the bandit self-improvement loop:
  kubectl exec -it deploy/{{ include "cogniflow.fullname" . }}-orchestrator -- \
    bandit-learn -log /data/bandit.jsonl -json -out /data/recommendation.json
  kubectl exec -it deploy/{{ include "cogniflow.fullname" . }}-orchestrator -- \
    kill -HUP 1  # orchestrator picks up the new recommendation on restart

4. (Optional) Persist usage events to Stripe:
  helm upgrade cogniflow ./deploy/cogniflow \
    --set secrets.stripeApiKey=$STRIPE_API_KEY \
    --set secrets.stripeMeterId=$STRIPE_METER_ID
